#include <iostream>


void swap(int* a, int* b) {
    int temp = *a;
    *a = *b;
    *b = temp;
}


int piv(int* arr, int low, int high) {
    int pivot = arr[high]; 
    int i = low - 1;      
    
    for (int j = low; j <= high - 1; j++) {
        if (arr[j] < pivot) {
            i++;
            swap(&arr[i], &arr[j]);
        }
    }
    swap(&arr[i + 1], &arr[high]);
    return (i + 1);
}


void quick_sort_2(int* arr, int low, int high) {
    if (low < high) {
        int pi = piv(arr, low, high); 

        quick_sort_2(arr, low, pi - 1);  
        quick_sort_2(arr, pi + 1, high); 
    }
}


void quick_sort(int* arr, int size) {
    quick_sort_2(arr, 0, size - 1);
}


void print_arr(int* arr, int size) {
    for (int i = 0; i < size; i++) {
        std::cout << arr[i] << " ";
    }
    std::cout << std::endl;
}

int main() {
    int arr[] = {88, 91, 87, 59, 53, 49, 29, 16, 4, 27, 28, 89, 2, 25, 74};
    int size = sizeof(arr) / sizeof(arr[0]);

    std::cout << "Исходный массив: ";
    print_arr(arr, size);

    quick_sort(arr, size);

    std::cout << "Отсортированный массив: ";
    print_arr(arr, size);

    return 0;
}
